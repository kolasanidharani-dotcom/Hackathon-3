
class Candidate {
    int rollNumber;
    String name;
    String position;
    int votes;

    // Parameterized constructor
    Candidate(int rollNumber, String name, String position, int votes) {
        this.rollNumber = rollNumber;
        this.name = name;
        this.position = position.toUpperCase();
        this.votes = votes;
    }

    // Method overloading
    void display() {
        display(true);
    }

    void display(boolean showVotes) {
        System.out.println("Candidate Registered Successfully");
        System.out.println("Candidate Name  : " + name.toUpperCase());
        System.out.println("Roll Number     : " + rollNumber);
        System.out.println("Position        : " + position);

        if (showVotes) {
            System.out.println("Votes Received  : " + votes);
        }
    }

    // Compare candidate names
    boolean isDuplicate(Candidate other) {
        return name.equalsIgnoreCase(other.name);
    }
}

public class Main {
    public static void main(String[] args) {
        Candidate c1 = new Candidate(101, "Ananya", "President", 125);
        Candidate c2 = new Candidate(102, "Rahul", "President", 98);
        Candidate c3 = new Candidate(103, "ANANYA", "President", 50);

        System.out.println("======= STUDENT COUNCIL ELECTION =======");

        c1.display();
        System.out.println();

        c2.display();
        System.out.println();

        System.out.println("Duplicate Candidate Check:");
        System.out.println("Candidate Name  : " + c3.name.toUpperCase());

        if (c1.isDuplicate(c3)) {
            System.out.println("Status          : Duplicate Nomination");
        } else {
            System.out.println("Status          : Valid Nomination");
        }

        // Find the winner for President
        Candidate winner = c1;

        if (c2.position.equals(c1.position) && c2.votes > winner.votes) {
            winner = c2;
        }

        System.out.println("\n---------- ELECTION RESULTS ----------");
        System.out.println("Position          : " + winner.position);
        System.out.println("Winning Candidate : " + winner.name.toUpperCase());
        System.out.println("Total Votes       : " + winner.votes);
        System.out.println("======================================");
    }
}
